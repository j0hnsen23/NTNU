# Classes, Promises

```javascript
class Book{
  constructor(name, author, isbn){
    this.name = name;
    this.author = author;
    this.isbn = isbn;
  }
  get description() {
    return `The book is called ${this.name} by ${this.author} and has ISBN: ${this.isbn}`;
  }
  set changeName(newName) {
    this.name = newName;
  }
}

const newBook = new Book("The Hunger Games", "Suzanne Collins", "978-0-439-02352-8");

console.log(newBook.description);

newBook.changeName = "Catching Fire";
console.log(newBook.description);


const myPromise = new Promise((resolve) => {
  setTimeout(() => {
    resolve("Data Loaded Successfully!");
  }, 3000);
});


myPromise
  .then((message) => {
    console.log(message); // skriver ut: "Data Loaded Successfully!"
  })
  .catch((error) => {
    console.log("Something went wrong:", error);
  });
```
